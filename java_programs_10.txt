4. Reverse String using Stack

import java.util.*;
public class ReverseString
{
public static void main(String ar[])
{
Scanner sc=new Scanner(System.in);
System.out.println("Enter any String:");
String str=sc.nextLine();
System.out.println("Original String:"+str);
Stack<Character> s=new Stack<>();
for(int i=0;i<str.length();i++)
s.push(str.charAt(i));
StringBuilder sb=new StringBuilder();
while(!s.isEmpty())
sb.append(s.pop());
System.out.println("Reversed String:"+sb.toString());
}
}

8. Binary Search

import java.util.*;
public class BinarySearch
{
public static void main(String ar[])
{
int i,s,n,l,h,j,mid,temp,index;
int[] a=new int[50];
Scanner sc=new Scanner(System.in);
System.out.println("Enter n:");
n=sc.nextInt();
System.out.println("Enter elements one by one:");
for(i=0;i<n;i++)
a[i]=sc.nextInt();
for(i=0;i<n;i++)
{
for(j=0;j<n;j++)
{
if(a[i]<a[j])
{
temp=a[i];
a[i]=a[j];
a[j]=temp;
}
}
}
System.out.println("\n Sorted elements");
for(i=0;i<n;i++)
System.out.println(a[i]+" ");
System.out.println("Enter the search elements:");
s=sc.nextInt();
l=0;h=n-1;index=-1;
while(l<=h)
{
mid=(l+h)/2;
if(a[mid]<s)
l=mid+1;
else if(a[mid]>s)
h=mid-1;
else
index=mid;
break;
}
if(index==-1)
System.out.println("Element is not found");
else
System.out.println("Element is found at:"+index);
}
}

9. Postfix Evaluation

import java.util.*;
public class Postfix
{
public static void main(String ar[])
{
String d="138*+";
System.out.println("Given Potfix expression:"+d);
Stack<Integer> s=new Stack<>();
for(char c:d.toCharArray())
{
if(Character.isDigit(c))
s.push(c-'0');
else
{
int x=s.pop();
int y=s.pop();
if(c=='+')
s.push(y+x);
if(c=='-')
s.push(y-x);
if(c=='*')
s.push(y*x);
if(c=='/')
s.push(y/x);
}
}
System.out.println("Result:"+s.pop());
}
}

10. Infix to Postfix

import java.util.*;
import java.lang.*;
public class InfixToPostfix
{
public static void main(String ar[])
{
Scanner sc=new Scanner(System.in);
System.out.println("Enter any Expression with paranthesis:");
String d= sc.nextLine();
System.out.println("Infix:"+d);
Stack<Character> s=new Stack<>();
StringBuilder sb=new StringBuilder();
int flag=0;
for(int i=0;i<d.length();i++)
{
char c=d.charAt(i);
if(Character.isLetterOrDigit(c))
sb.append(c);
else if(c=='+'|| c=='-'|| c=='*' || c=='/' || c=='(')
s.push(c);
else
{
while(!s.isEmpty() && s.peek()!= '(')
{
sb.append(s.pop());
flag=1;
}
if(flag==1)
s.pop();
flag=0;
}
}
System.out.println("Postfix:"+sb.toString());
}
}
